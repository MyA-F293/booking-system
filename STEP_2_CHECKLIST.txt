📋 PERSIAPAN
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
[ ] Buka 2 tab browser:
    Tab 1: Supabase Dashboard (https://supabase.com/dashboard)
    Tab 2: GitHub Repo Settings

[ ] Siapkan notepad untuk catat 3 values

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
🔑 DAPAT SECRET #1: SUPABASE_URL
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
[ ] 1. Di Supabase Dashboard
[ ] 2. Pilih project booking-system
[ ] 3. Klik Settings (icon gear di sidebar kiri bawah)
[ ] 4. Klik API
[ ] 5. Copy "Project URL"
    Contoh: https://abcdefghijk.supabase.co
[ ] 6. Paste ke notepad, label: SUPABASE_URL

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
🔑 DAPAT SECRET #2: SUPABASE_ANON_KEY
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
[ ] 1. Masih di halaman Settings > API
[ ] 2. Scroll ke "Project API keys"
[ ] 3. Cari yang namanya "anon" "public"
[ ] 4. Klik icon Copy
    (Key panjang, dimulai: eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...)
[ ] 5. Paste ke notepad, label: SUPABASE_ANON_KEY

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
🔑 DAPAT SECRET #3: SUPABASE_API_KEY
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
[ ] 1. Ini API key yang ANDA BUAT di Step 1
    Contoh: eysa-booking-secret-2026-abc123
[ ] 2. Copy dari notepad Step 1
[ ] 3. Paste ke notepad, label: SUPABASE_API_KEY

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
🚀 TAMBAH KE GITHUB
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
[ ] 1. Buka GitHub repo booking-system
[ ] 2. Klik tab "Settings"
[ ] 3. Sidebar kiri: "Secrets and variables" > "Actions"
[ ] 4. Klik "New repository secret"

[ ] 5. Tambah Secret #1:
    Name:   SUPABASE_URL
    Secret: [paste dari notepad]
    Klik "Add secret"

[ ] 6. Klik "New repository secret" lagi

[ ] 7. Tambah Secret #2:
    Name:   SUPABASE_ANON_KEY
    Secret: [paste dari notepad]
    Klik "Add secret"

[ ] 8. Klik "New repository secret" lagi

[ ] 9. Tambah Secret #3:
    Name:   SUPABASE_API_KEY
    Secret: [paste dari notepad]
    Klik "Add secret"
